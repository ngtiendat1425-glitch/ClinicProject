function App() {
  const [activeTab, setActiveTab] = useState<Tab>('home');
  const [search, setSearch] = useState('');
  const [specialty, setSpecialty] = useState('Tất cả');
  const [selectedDoctor, setSelectedDoctor] = useState<Doctor | null>(null);
  const [selectedDate, setSelectedDate] = useState(dates[0]);
  const [selectedTime, setSelectedTime] = useState(times[1]);
  const [appointments, setAppointments] = useState<Appointment[]>([
    {
      id: 'sample',
      doctor: doctors[0],
      date: 'Thứ Tư, 18 tháng 6',
      time: '09:30',
      type: 'Khám trực tiếp',
      status: 'Đã xác nhận',
    },
  ]);

  const filteredDoctors = useMemo(() => {
    const query = search.trim().toLocaleLowerCase('vi');
    return doctors.filter((doctor) => {
      const matchesSpecialty = specialty === 'Tất cả' || doctor.specialty === specialty;
      const matchesSearch =
        query.length === 0 ||
        doctor.name.toLocaleLowerCase('vi').includes(query) ||
        doctor.specialty.toLocaleLowerCase('vi').includes(query);
      return matchesSpecialty && matchesSearch;
    });
  }, [search, specialty]);

  const openBooking = (doctor: Doctor) => {
    setSelectedDoctor(doctor);
    setSelectedDate(dates[0]);
    setSelectedTime(times[1]);
  };

  const confirmBooking = () => {
    if (!selectedDoctor) return;
    setAppointments((current) => [
      {
        id: `${Date.now()}`,
        doctor: selectedDoctor,
        date: `${selectedDate.day}, ${selectedDate.date} tháng 6`,
        time: selectedTime,
        type: 'Khám trực tiếp',
        status: 'Chờ xác nhận',
      },
      ...current,
    ]);
    setSelectedDoctor(null);
    setActiveTab('appointments');
    Alert.alert('Đặt lịch thành công', 'Phòng khám sẽ xác nhận lịch hẹn của bạn sớm nhất.');
  };

  const cancelAppointment = (id: string) => {
    Alert.alert('Hủy lịch hẹn?', 'Bạn có chắc muốn hủy lịch hẹn này không?', [
      { text: 'Giữ lịch', style: 'cancel' },
      {
        text: 'Hủy lịch',
        style: 'destructive',
        onPress: () =>
          setAppointments((current) =>
            current.map((appointment) =>
              appointment.id === id ? { ...appointment, status: 'Đã hủy' } : appointment,
            ),
          ),
      },
    ]);
  };

  return (
    <View style={styles.app}>
      <StatusBar style="dark" />
      <NativeStatusBar barStyle="dark-content" backgroundColor={colors.canvas} />
      <View style={styles.screen}>
        {activeTab === 'home' && (
          <HomeScreen
            search={search}
            onSearch={setSearch}
            onSelectSpecialty={(value) => {
              setSpecialty(value);
              setActiveTab('doctors');
            }}
            onSeeDoctors={() => setActiveTab('doctors')}
            onBook={openBooking}
          />
        )}
        {activeTab === 'doctors' && (
          <DoctorsScreen
            search={search}
            onSearch={setSearch}
            specialty={specialty}
            onSelectSpecialty={setSpecialty}
            doctors={filteredDoctors}
            onBook={openBooking}
          />
        )}
        {activeTab === 'appointments' && (
          <AppointmentsScreen appointments={appointments} onCancel={cancelAppointment} />
        )}
        {activeTab === 'profile' && <ProfileScreen />}
      </View>
      <TabBar activeTab={activeTab} onChange={setActiveTab} />

      <Modal
        visible={selectedDoctor !== null}
        transparent
        animationType="slide"
        onRequestClose={() => setSelectedDoctor(null)}
      >
        <View style={styles.modalBackdrop}>
          <Pressable style={StyleSheet.absoluteFill} onPress={() => setSelectedDoctor(null)} />
          <View style={styles.bookingSheet}>
            <View style={styles.sheetHandle} />
            <View style={styles.sheetHeading}>
              <View>
                <Text style={styles.eyebrow}>ĐẶT LỊCH KHÁM</Text>
                <Text style={styles.sheetTitle}>Chọn thời gian</Text>
              </View>
              <Pressable onPress={() => setSelectedDoctor(null)} style={styles.closeButton}>
                <Text style={styles.closeText}>×</Text>
              </Pressable>
            </View>
            {selectedDoctor && <DoctorCompact doctor={selectedDoctor} />}
            <Text style={styles.sectionLabel}>Ngày khám</Text>
            <ScrollView horizontal showsHorizontalScrollIndicator={false}>
              <View style={styles.dateRow}>
                {dates.map((item) => {
                  const selected = selectedDate.date === item.date;
                  return (
                    <Pressable
                      key={item.date}
                      style={[styles.dateCard, selected && styles.dateCardSelected]}
                      onPress={() => setSelectedDate(item)}
                    >
                      <Text style={[styles.dateDay, selected && styles.selectedText]}>{item.day}</Text>
                      <Text style={[styles.dateNumber, selected && styles.selectedText]}>
                        {item.date}
                      </Text>
                      <Text style={[styles.dateMonth, selected && styles.selectedSubText]}>Tháng 6</Text>
                    </Pressable>
                  );
                })}
              </View>
            </ScrollView>
            <Text style={styles.sectionLabel}>Giờ khám</Text>
            <View style={styles.timeGrid}>
              {times.map((time) => {
                const selected = selectedTime === time;
                return (
                  <Pressable
                    key={time}
                    style={[styles.timeChip, selected && styles.timeChipSelected]}
                    onPress={() => setSelectedTime(time)}
                  >
                    <Text style={[styles.timeText, selected && styles.selectedText]}>{time}</Text>
                  </Pressable>
                );
              })}
            </View>
            <View style={styles.bookingTotal}>
              <Text style={styles.totalLabel}>Phí khám dự kiến</Text>
              <Text style={styles.totalPrice}>{selectedDoctor?.fee}</Text>
            </View>
            <PrimaryButton title="Xác nhận đặt lịch" onPress={confirmBooking} />
            <Text style={styles.sheetNote}>Bạn chưa cần thanh toán ở bước này</Text>
          </View>
        </View>
      </Modal>
    </View>
  );
}

function AppointmentsScreen({
  appointments,
  onCancel,
}: {
  appointments: Appointment[];
  onCancel: (id: string) => void;
}) {
  const upcoming = appointments.filter((item) => item.status !== 'Đã hủy');
  const history = appointments.filter((item) => item.status === 'Đã hủy');
  return (
    <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={styles.scrollContent}>
      <View style={styles.pageHeader}>
        <Text style={styles.pageEyebrow}>THEO DÕI SỨC KHỎE</Text>
        <Text style={styles.pageTitle}>Lịch hẹn</Text>
        <Text style={styles.pageSubtitle}>Quản lý các buổi khám của bạn</Text>
      </View>
      <View style={styles.appointmentSummary}>
        <View style={styles.summaryIcon}><Text style={styles.summaryIconText}>▣</Text></View>
        <View style={styles.summaryContent}>
          <Text style={styles.summaryTitle}>{upcoming.length} lịch hẹn sắp tới</Text>
          <Text style={styles.summarySubtitle}>Đừng quên lịch khám của bạn nhé</Text>
        </View>
        <Text style={styles.summaryArrow}>›</Text>
      </View>
      <Text style={styles.sectionTitle}>Sắp tới</Text>
      {upcoming.length > 0 ? (
        upcoming.map((appointment) => (
          <AppointmentCard key={appointment.id} appointment={appointment} onCancel={onCancel} />
        ))
      ) : (
        <View style={styles.emptyState}>
          <Text style={styles.emptyIcon}>▣</Text>
          <Text style={styles.emptyTitle}>Bạn chưa có lịch hẹn</Text>
          <Text style={styles.emptyCopy}>Lịch khám mới sẽ hiển thị tại đây.</Text>
        </View>
      )}
      {history.length > 0 && (
        <>
          <Text style={[styles.sectionTitle, styles.historyHeading]}>Lịch đã hủy</Text>
          {history.map((appointment) => <AppointmentCard key={appointment.id} appointment={appointment} />)}
        </>
      )}
      <View style={styles.helpCard}>
        <Text style={styles.helpIcon}>i</Text>
        <View style={styles.helpTextBlock}>
          <Text style={styles.helpTitle}>Cần hỗ trợ?</Text>

        </View>
        <Text style={styles.helpArrow}>›</Text>
      </View>
      <View style={styles.bottomSpace} />
    </ScrollView>
  );
}

          <Text style={styles.appointmentDate}>{appointment.date}</Text>

        <View style={[styles.statusPill, { backgroundColor: statusBackground }]}>
          <Text style={[styles.statusText, { color: statusColor }]}>{appointment.status}</Text>
        </View>
      </View>
      <View style={styles.appointmentDivider} />

          <Text style={styles.smallAvatarText}>{appointment.doctor.initials}</Text>
        </View>
        <View style={styles.doctorInfo}>
          <Text style={styles.doctorName}>{appointment.doctor.name}</Text>
          <Text style={styles.doctorSpecialty}>{appointment.doctor.specialty}  ·  {appointment.type}</Text>
        </View>
      </View>

    </View>

}

function TabBar({ activeTab, onChange }: { activeTab: Tab; onChange: (tab: Tab) => void }) {
  const tabs: { id: Tab; label: string; icon: string }[] = [
    { id: 'home', label: 'Trang chủ', icon: '⌂' },
    { id: 'doctors', label: 'Bác sĩ', icon: '✚' },
    { id: 'appointments', label: 'Lịch hẹn', icon: '▣' },
    { id: 'profile', label: 'Cá nhân', icon: '◉' },
  ];
  return (
    <View style={styles.tabBar}>
      {tabs.map((tab) => {
        const selected = activeTab === tab.id;
        return (
          <Pressable key={tab.id} style={styles.tabItem} onPress={() => onChange(tab.id)}>
            <View style={[styles.tabIconWrap, selected && styles.tabIconWrapSelected]}>
              <Text style={[styles.tabIcon, selected && styles.tabIconSelected]}>{tab.icon}</Text>
            </View>
            <Text style={[styles.tabLabel, selected && styles.tabLabelSelected]}>{tab.label}</Text>
          </Pressable>
        );
      })}
    </View>
  );
}
